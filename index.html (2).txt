<script>

/* GAME VARIABLES */

let balance = 10000;

let pot = 1200;

let gameOver = false;


/* UPDATE */

function update(){

    document.getElementById("balance")
        .innerText =
        balance.toLocaleString();


    document.getElementById("playerStack")
        .innerText =
        balance.toLocaleString();


    document.getElementById("pot")
        .innerText =
        pot.toLocaleString();

}


/* STATUS */

function setStatus(text,color){

    const status =
        document.getElementById("status");

    status.innerText = text;

    status.style.color = color;

}


/* FOLD */

function fold(){

    if(gameOver) return;

    gameOver = true;

    setStatus(
        "YOU FOLDED",
        "#ff6672"
    );

}


/* CHECK */

function check(){

    if(gameOver) return;

    setStatus(
        "CHECK — OPPONENT THINKING...",
        "#9ee6c4"
    );

}


/* RAISE */

function raiseBet(){

    if(gameOver) return;

    const amount = 500;


    if(balance < amount){

        setStatus(
            "NOT ENOUGH CHIPS",
            "#ff6672"
        );

        return;

    }


    balance -= amount;

    pot += amount;


    setStatus(
        "SANCHIR RAISES 500",
        "#ffd866"
    );


    update();

}


/* NEW GAME */

function newGame(){

    balance = 10000;

    pot = 1200;

    gameOver = false;


    setStatus(
        "NEW HAND — YOUR TURN",
        "#ffffff"
    );


    update();

}


/* START */

update();

</script>
